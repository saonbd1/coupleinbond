<script>
const params = new URLSearchParams(location.search);
const mode = params.get('mode');
const out = document.getElementById('out');

if (mode === 'self') {
  // Report the top-level viewport metrics for the current window.
  const vw = document.documentElement.clientWidth;
  const lines = [];
  lines.push('innerWidth: ' + window.innerWidth);
  lines.push('clientWidth: ' + vw);
  lines.push('devicePixelRatio: ' + window.devicePixelRatio);
  lines.push('document scrollWidth: ' + document.documentElement.scrollWidth);
  lines.push('horizontal overflow: ' + (document.documentElement.scrollWidth > vw));
  const offenders = [];
  document.querySelectorAll('*').forEach((el) => {
    const r = el.getBoundingClientRect();
    if (r.width > vw + 1 || r.right > vw + 1) {
      offenders.push(el.tagName.toLowerCase() + '.' + (el.className || '') +
        '  w=' + Math.round(r.width) + ' right=' + Math.round(r.right));
    }
  });
  lines.push('--- overflowing elements (' + offenders.length + ') ---');
  offenders.slice(0, 25).forEach((l) => lines.push(l));
  out.textContent = lines.join('\n');
} else {
  // Same-origin probe: loads a target page in a fixed-width iframe and reports
  // the document scroll width plus the elements that overflow the viewport.
  const target = params.get('t') || 'blog.html';
  const width = Number(params.get('w') || 390);
  const f = document.createElement('iframe');
  f.style.width = width + 'px';
  f.style.height = '800px';
  f.style.border = '0';
  document.body.appendChild(f);

  f.addEventListener('load', () => {
    const doc = f.contentDocument;
    const vw = doc.documentElement.clientWidth;
    const lines = [];
    lines.push('target: ' + target);
    lines.push('iframe width: ' + width);
    lines.push('viewport width: ' + vw);
    lines.push('document scrollWidth: ' + doc.documentElement.scrollWidth);
    lines.push('horizontal overflow: ' + (doc.documentElement.scrollWidth > vw));

    const offenders = [];
    doc.querySelectorAll('*').forEach((el) => {
      const r = el.getBoundingClientRect();
      if (r.width > vw + 1 || r.right > vw + 1) {
        offenders.push({
          tag: el.tagName.toLowerCase(),
          cls: (el.className && el.className.baseVal !== undefined ? el.className.baseVal : el.className) || '',
          width: Math.round(r.width),
          right: Math.round(r.right),
        });
      }
    });
    lines.push('--- overflowing elements (' + offenders.length + ') ---');
    offenders.slice(0, 25).forEach((o) => {
      lines.push(o.tag + '.' + o.cls + '  w=' + o.width + ' right=' + o.right);
    });
    out.textContent = lines.join('\n');
  });
  f.src = target;
}
</script>
